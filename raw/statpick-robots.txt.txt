# Machine-readable entry points for AI agents and crawlers:
#   https://www.statpick.ai/llms.txt      — plain-language site guide
#   https://www.statpick.ai/openapi.json  — OpenAPI 3.1 spec for the public API
#   https://www.statpick.ai/sitemap.xml   — sitemap index (per-surface maps below)

# Block only routes that genuinely shouldn't be crawled. Auth pages like
# /log-in, /register, /get-started, and personalized surfaces like
# /my-picks, /favorites, /activity, /profile stay crawlable so Google can
# rank them and consider them as sitelink candidates on branded queries.
Disallow: /admin
Disallow: /r/
Disallow: /api/

# --- AI assistant crawlers -------------------------------------------------
# GPTBot and ClaudeBot are the crawlers behind ChatGPT and Claude. They get the
# same access as any search engine, programmatic prop/player/team pages
# included: being citable in an AI answer is worth more than the crawl cost,
# and a thin corpus is exactly what makes an assistant answer about us from
# somebody else's page.
#
# COST NOTE — this is the trade being made. /props, /players, /teams, /games,
# /box-scores, /stats, /nrfi, and /start-sit are tens of thousands of ISR
# pages, and every crawler hit on an expired one is a Worker render plus an
# R2 write. The public API (see /openapi.json) serves the same numbers with
# no render cost, which is why llms.txt points agents there first. If crawl
# spend becomes a problem, throttle at Cloudflare rather than re-closing this
# door — a rate limit slows a crawl, a Disallow deletes us from the corpus.
User-agent: GPTBot
User-agent: ClaudeBot
Allow: /
Disallow: /admin
Disallow: /r/
Disallow: /api/
# Advisory pacing. Cloudflare's Free plan can't rate-limit on User-Agent, so
# this is the only crawl-rate lever available; support is inconsistent across
# crawlers, so treat it as a request, not a control. ~2s => ~43k pages/day if
# honoured, which covers a full crawl of the programmatic surface in about a
# day without hammering it.
Crawl-delay: 2

# Crawl feeds a large share of LLM training corpora, so it buys corpus reach
# the others don't, and it is allowed under the `*` group above. Search bots (Googlebot, Bingbot) and
