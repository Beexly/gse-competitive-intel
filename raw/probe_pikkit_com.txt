      $('a#store').attr('href', playStoreLink);
      $('storesticky').attr('href', playStoreLink);
    }

  // Update the store buttons on page load
  updateStoreButtons();
  
  // Add a click event listener to the store buttons
  $('button#storebtn, button#store, button#storesticky,').click(function() {
    if (mode === 'appstore') {
      mode = 'playstore';
    } else if (mode === 'playstore') {
      mode = 'appstore';
    }
    updateStoreButtons();
  });
});
