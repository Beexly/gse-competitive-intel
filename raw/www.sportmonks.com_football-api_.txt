    /* outer tabs: Match Centre / See the data behind it */
    var otabs = card.querySelectorAll('.sm-fah3-otab');
    var opanels = card.querySelectorAll('.sm-fah3-opanel');
    otabs.forEach(function(tab){
      tab.addEventListener('click', function(){
        var name = tab.getAttribute('data-otab');
        otabs.forEach(function(t){ t.classList.toggle('is-active', t === tab); });
        opanels.forEach(function(p){ p.classList.toggle('is-active', p.getAttribute('data-opanel') === name); });
      });
    });

    /* inner match-centre tabs: Events / Stats / Predictions */
    var tabs = card.querySelectorAll('.sm-fah3-tab');
    var panels = card.querySelectorAll('.sm-fah3-panel');
    function activate(name){
      tabs.forEach(function(t){ t.classList.toggle('is-active', t.getAttribute('data-tab') === name); });
      panels.forEach(function(p){ p.classList.toggle('is-active', p.getAttribute('data-panel') === name); });
    }
    tabs.forEach(function(tab){ tab.addEventListener('click', function(){ activate(tab.getAttribute('data-tab')); }); });
  });
})();
</script>

(function(){

  function buildDots(){
    dotsWrap.innerHTML = ''; dots = [];

      dotsWrap.appendChild(d); dots.push(d);
    }
    syncDots();
  }
  function syncDots(){
    var idx = Math.round(track.scrollLeft / cardStep());
    dots.forEach(function(d, i){ d.classList.toggle('is-on', i === Math.min(idx, dots.length - 1)); });
  }
  track.addEventListener('scroll', function(){ window.requestAnimationFrame(syncDots); });
  window.addEventListener('resize', buildDots);
  document.getElementById('sm-fatest-prev').addEventListener('click', function(){ stop(); track.scrollBy({ left: -cardStep(), behavior: reduced ? 'auto' : 'smooth' }); });
  document.getElementById('sm-fatest-next').addEventListener('click', function(){ stop(); track.scrollBy({ left: cardStep(), behavior: reduced ? 'auto' : 'smooth' }); });
  var timer = null;
  function start(){
    if (reduced || timer) return;
    timer = setInterval(function(){
      var max = track.scrollWidth - track.clientWidth;
      var next = track.scrollLeft >= max - 10 ? 0 : track.scrollLeft + cardStep();
      track.scrollTo({ left: next, behavior: 'smooth' });
    }, 5000);
  }

  buildDots();
  if ('IntersectionObserver' in window) {
    var io = new IntersectionObserver(function(es){ es.forEach(function(e){ if (e.isIntersecting) start(); else stop(); }); }, { threshold: 0.3 });
