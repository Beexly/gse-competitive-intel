<!-- Membership hierarchy management -->
<script>
(function() {
    const API_BASE = '/backend/wp-json/deepbetting/v1';

    // Textes pour le bouton désactivé (i18n)
    const disabledTexts = {
        en: 'Current Plan',
        fr: 'Abonnement actuel',
        es: 'Plan actual',
        pt: 'Plano atual'
    };

    async function fetchMembershipHierarchy() {
        try {
            const response = await fetch(API_BASE + '/memberships-hierarchy', {
                credentials: 'include'
            });
            if (!response.ok) return null;
            return await response.json();
        } catch (e) {
            console.error('Error fetching membership hierarchy:', e);
            return null;
        }
    }

    function getDisabledText() {
        const lang = localStorage.getItem('deepbetting_language') || 'en';
        return disabledTexts[lang] || disabledTexts['en'];
    }

    function disableButton(btn) {
        btn.classList.add('btn-disabled');
        btn.setAttribute('aria-disabled', 'true');
        btn.removeAttribute('href');
        btn.style.pointerEvents = 'none';
        
        // Changer le texte du bouton
        const textSpan = btn.querySelector('span');
        if (textSpan) {
            textSpan.textContent = getDisabledText();
        } else {
            btn.textContent = getDisabledText();
        }
        
        // Ajouter une icône de check
        if (!btn.querySelector('.fa-check')) {
            const icon = document.createElement('i');
            icon.className = 'fas fa-check';
            icon.style.marginRight = '8px';
            btn.prepend(icon);
        }
    }

    function disableCard(card) {
        card.classList.add('pricing-card-disabled');
    }

    async function applyMembershipRestrictions() {
        const data = await fetchMembershipHierarchy();
        
        if (!data || !data.logged_in || data.user_level === -1) {
            // Non connecté ou pas d'abonnement : ne rien bloquer
            return;
        }

        const blockedIds = data.blocked || [];
        
        // Désactiver les boutons des memberships bloqués
        blockedIds.forEach(membershipId => {
            const btn = document.querySelector(`.pricing-cta[data-membership-id="${membershipId}"]`);
            const card = document.querySelector(`.pricing-card[data-membership-id="${membershipId}"]`);
            
            if (btn) disableButton(btn);
            if (card) disableCard(card);
        });

        // Marquer les memberships actuels de l'utilisateur
        const userMemberships = data.user_memberships || [];
        userMemberships.forEach(membershipId => {
            const card = document.querySelector(`.pricing-card[data-membership-id="${membershipId}"]`);
            if (card) {
                card.classList.add('pricing-card-current');
            }
        });
    }

    // Exécuter après le chargement du DOM
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', applyMembershipRestrictions);
    } else {
        applyMembershipRestrictions();
    }

    // Ré-appliquer si la langue change (pour mettre à jour le texte du bouton)
    document.querySelectorAll('.lang-option').forEach(option => {
        option.addEventListener('click', function() {
            setTimeout(() => {
                document.querySelectorAll('.btn-disabled').forEach(btn => {
                    const textSpan = btn.querySelector('span');
                    if (textSpan) {
                        textSpan.textContent = getDisabledText();
                    } else if (!btn.querySelector('.fa-check')) {
                        btn.textContent = getDisabledText();
                    } else {
                        // Préserver l'icône
                        const icon = btn.querySelector('.fa-check');
                        btn.textContent = getDisabledText();
                        btn.prepend(icon);
                    }
                });
            }, 100);
        });
    });
})();
</script>

<script>
document.addEventListener('DOMContentLoaded', function() {
    const flagUrls = {
        en: './assets/img/languages/gb.svg',
        fr: './assets/img/languages/fr.svg',
        es: './assets/img/languages/es.svg',
        pt: './assets/img/languages/br.svg'
    };

    // Override the global flagUrls if i18n.js defines it
    if (typeof window.flagUrls !== 'undefined') {
        window.flagUrls = flagUrls;
    }

    // Function to update the current flag
    function updateCurrentFlag() {
        const currentLang = localStorage.getItem('deepbetting_language') || 'en';
        const currentFlag = document.getElementById('currentFlag');
        if (currentFlag && flagUrls[currentLang]) {
            currentFlag.src = flagUrls[currentLang];
        }
    }

    // Update flag on page load
    updateCurrentFlag();

    // Listen for language changes
    document.querySelectorAll('.lang-option').forEach(option => {
        option.addEventListener('click', function() {
            const lang = this.getAttribute('data-lang');
            const currentFlag = document.getElementById('currentFlag');
            if (currentFlag && flagUrls[lang]) {
                currentFlag.src = flagUrls[lang];
            }
        });
    });
});
</script>
